    ```typescript
    const abortRef = useRef<AbortController | null>(null);

    const fetchData = useCallback(async () => {
      abortRef.current?.abort();                      // cancel prior request (REL-05)
      const controller = new AbortController();
      abortRef.current = controller;
      setLoading(true);
      setError(null);
      try {
        const token = await getAccessToken();
        const res = await fetch(`${API}/api/...`, {
          headers: token ? { Authorization: `Bearer ${token}` } : {},
          signal: controller.signal,                  // wire abort
        });
        const json = await res.json();
        if (!res.ok) {
          setError((json as { detail?: string; error?: string }).detail ?? json.error ?? "Erro ao carregar dados.");
          return;
        }
        setData(json);
      } catch (err) {
        if ((err as Error).name === "AbortError") return;  // cancelled — ignore (REL-05)
        setError("Erro de conexão com o servidor.");
      } finally {
        if (!controller.signal.aborted) setLoading(false);
      }
    }, []);

    useEffect(() => {
      fetchData();
      return () => abortRef.current?.abort();          // cleanup on unmount (StrictMode safe)
    }, [fetchData]);
    ```
